  const item = practiceItems[pIndex];
  document.getElementById('pzLabel').textContent = `§ 01 · Practise — item ${pIndex+1} of ${practiceItems.length}`;
  document.getElementById('pzQ').textContent = item.q;
  pSelected=null; pSubmitted=false;

  opts.innerHTML='';
  item.opts.forEach((text,i)=>{
    const b=document.createElement('button');
    b.className='pz-opt'; b.type='button'; b.role='radio'; b.setAttribute('aria-checked','false');
    b.tabIndex = i===0 ? 0 : -1;

function moveFocus(i,dir){
  const opts=[...document.querySelectorAll('.pz-opt')];
  const next = opts[(i+dir+opts.length)%opts.length];
  opts.forEach(o=>o.tabIndex=-1);
  next.tabIndex=0; next.focus();
}
function selectOpt(i){
  if(pSubmitted) return;
  pSelected=i;
  document.querySelectorAll('.pz-opt').forEach((o,idx)=>o.setAttribute('aria-checked', idx===i ? 'true':'false'));
  document.getElementById('submitBtn').disabled=false;
}

    if(pIndex < practiceItems.length-1){ pIndex++; renderPractice(); announce('Next item loaded.'); }
    return;

  if(pSelected===null) return;
  pSubmitted=true;
  const item = practiceItems[pIndex];
  const correct = pSelected === item.correct;
  document.querySelectorAll('.pz-opt').forEach((o,idx)=>{
    o.disabled=true;
    if(idx===item.correct) o.classList.add('correct');
    else if(idx===pSelected) o.classList.add('incorrect');
  });
  const stamp = document.getElementById('pStamp');
  stamp.className = 'pz-stamp show ' + (correct?'ok':'bad');
  stamp.textContent = correct
    ? '✓ recorded · correct · ratio estimate moved +0.03'
    : '✗ recorded · not quite · see worked example next time · estimate adjusted −0.01';
  const btn = document.getElementById('submitBtn');
  if(pIndex < practiceItems.length-1){ btn.textContent='Next item →'; btn.disabled=false; }
  else { btn.textContent='Done'; btn.disabled=true; }
  announce(correct ? 'Answer recorded as correct.' : 'Answer recorded as incorrect.');
});

const frames = [
  {label:'01 open', title:'Setting the scene', body:'A tank fills at a constant rate. Before pressing play: what do you expect the water level to look like after 10 seconds?', actions:['Predict','Change variables','Ask why']},
  {label:'02 predict', title:'Your prediction', body:'You predicted a straight line. Press play to see the actual fill curve and compare.', actions:['Play','Compare to prediction']},
  {label:'03 vary', title:'Change a variable', body:'Try doubling the inflow rate. What happens to the time to fill?', actions:['Double inflow','Halve inflow','Reset']},
  {label:'04 explain', title:'Ask why', body:'The curve is a straight line because the rate is constant — equal time steps add equal volume. Ask the tutor for a different explanation.', actions:['Ask differently','Resume']}
];
let activeFrame=0;
function renderFilm(){
  const film = document.getElementById('film');
  film.innerHTML='';
  frames.forEach((f,i)=>{
    const b=document.createElement('button');
    b.className='frame'; b.setAttribute('role','tab');
    b.setAttribute('aria-selected', i===activeFrame ? 'true':'false');
    b.tabIndex = i===activeFrame ? 0 : -1;
    b.textContent=f.label;
    b.addEventListener('click', ()=>{ activeFrame=i; renderFilm(); renderFrameMain(); });
    b.addEventListener('keydown', e=>{
      if(e.key==='ArrowRight'){ e.preventDefault(); activeFrame=(i+1)%frames.length; renderFilm(); renderFrameMain(); film.children[activeFrame].focus(); }
      if(e.key==='ArrowLeft'){ e.preventDefault(); activeFrame=(i-1+frames.length)%frames.length; renderFilm(); renderFrameMain(); film.children[activeFrame].focus(); }
    });
    film.appendChild(b);
  });
}
function renderFrameMain(){
  const f = frames[activeFrame];
  document.getElementById('frameMain').innerHTML = `
    <h4>Frame ${activeFrame+1} — ${f.title}</h4>
    <p>${f.body}</p>
    <div class="frame-actions">${f.actions.map(a=>`<button class="fa-btn">${a}</button>`).join('')}</div>
  `;
}
renderFilm(); renderFrameMain();
